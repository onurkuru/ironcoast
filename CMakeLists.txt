cmake_minimum_required(VERSION 3.16)
if(DEFINED ENV{VITASDK} AND KH_VITA)
  set(CMAKE_TOOLCHAIN_FILE "$ENV{VITASDK}/share/vita.toolchain.cmake" CACHE PATH "Vita toolchain")
endif()
project(KiyiHurdasi VERSION 0.2.3 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(kh_core src/game.cpp src/campaign.cpp)
target_include_directories(kh_core PUBLIC src)
target_compile_options(kh_core PRIVATE -Wall -Wextra -Wpedantic)
if(KH_VITA)
 include("$ENV{VITASDK}/share/vita.cmake" REQUIRED)
 add_executable(kiyi_hurdasi src/main.cpp src/render.cpp src/audio.cpp)
 target_compile_definitions(kiyi_hurdasi PRIVATE KH_VITA)
 target_include_directories(kiyi_hurdasi PRIVATE "$ENV{VITASDK}/arm-vita-eabi/include/SDL2" third_party)
 target_link_libraries(kiyi_hurdasi kh_core SDL2 SceGxm_stub SceDisplay_stub SceCtrl_stub SceAppMgr_stub SceAppUtil_stub SceAudio_stub SceAudioIn_stub SceSysmodule_stub SceIofilemgr_stub SceCommonDialog_stub SceTouch_stub SceHid_stub SceMotion_stub ScePower_stub SceProcessmgr_stub m)
 vita_create_self(eboot.bin kiyi_hurdasi SAFE)
 vita_create_vpk(kiyi-hurdasi.vpk KHYI00001 eboot.bin VERSION 00.23 NAME "Iron Coast: Scrap Tide" FILE assets assets)
else()
 find_package(PkgConfig REQUIRED)
 pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
 add_executable(kiyi_hurdasi src/main.cpp src/render.cpp src/audio.cpp)
 target_include_directories(kiyi_hurdasi PRIVATE third_party)
 target_link_libraries(kiyi_hurdasi kh_core PkgConfig::SDL2)
 target_compile_options(kiyi_hurdasi PRIVATE -Wall -Wextra -Wpedantic)
 add_executable(kh_tests tests/test_game.cpp)
 target_link_libraries(kh_tests kh_core)
 enable_testing()
 add_test(NAME campaign_and_gameplay COMMAND kh_tests)
 add_executable(kh_render_tests tests/test_render.cpp src/render.cpp)
 target_include_directories(kh_render_tests PRIVATE third_party)
 target_link_libraries(kh_render_tests kh_core PkgConfig::SDL2)
 add_test(NAME rendering_and_scene_changes COMMAND kh_render_tests "${CMAKE_CURRENT_SOURCE_DIR}/assets")
endif()
